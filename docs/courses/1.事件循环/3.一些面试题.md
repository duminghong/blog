```jvascript

function delay(duration = 0) {
  return new Promise((resolve) => {
    setTimeout(resolve, duration)
  })
};

delay().then(()=> {
  console.log(1);
  Promise.resolve()
    .then(() => {
      console.log(2);
    })
})

Promise.resolve()
  .then(delay)
  .then(() => {
    console.log(3);
  })

console.log(4);

```
